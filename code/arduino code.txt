/**
 * @file Guardian_Robot_Complete.ino
 * @brief Complete Guardian Robot - All Sensors + Line Following
 * 
 * This code implements a fully integrated guardian robot with multiple sensors
 * and operational modes. The robot features manual control, automatic line
 * following, obstacle detection, gas leak detection, sound detection,
 * voice alerts, and headlight control.
 * 
 * ## Features
 * - **Manual Mode**: Full Bluetooth car control (F, B, L, R, S)
 * - **Auto Line Following Mode**: Robot follows black line automatically
 * - **Ultrasonic Sensor**: Auto stop & buzzer when obstacle detected
 * - **Gas Sensor (A0)**: Voice alert + LED blink + buzzer when gas detected
 * - **Sound Sensor (A5)**: Voice alert + LED blink + buzzer when loud sound detected
 * - **Voice Horn**: ISD1820 + buzzer when Y pressed
 * - **Headlights**: On/Off control with X/x
 * 
 * ## Priority System
 * GAS > SOUND > ULTRASONIC > MODE OPERATION
 * 
 * ## Pin Configuration
 * | Component        | Pins        | Description                    |
 * |------------------|-------------|--------------------------------|
 * | L298N Motor      | ENA=3       | Left motor speed (PWM)        |
 * |                  | IN1=4, IN2=5| Left motor direction          |
 * |                  | IN3=6, IN4=7| Right motor direction         |
 * |                  | ENB=8       | Right motor speed (PWM)       |
 * | TCRT5000 Sensors | 11, 12      | Left/Right line sensors       |
 * | HC-SR04          | 9, 10       | Trigger/Echo pins             |
 * | MQ-2 Gas Sensor  | A0          | Analog output                 |
 * | KY-037 Sound     | A5          | Analog output                 |
 * | ISD1820 Voice    | A3          | Trigger pin                   |
 * | Headlights       | A1, A2      | LED outputs                   |
 * | Buzzer           | A4          | Active buzzer                 |
 * 
 * @author Guardian Robot Project
 * @version 1.0.0
 */

// ============================================
//                          PIN DEFINITIONS
// ============================================

// Motor Control (L298N Dual H-Bridge)
#define ENA 3   /**< Left motor speed (PWM pin) */
#define IN1 4   /**< Left motor direction 1 */
#define IN2 5   /**< Left motor direction 2 */
#define IN3 6   /**< Right motor direction 1 */
#define IN4 7   /**< Right motor direction 2 */
#define ENB 8   /**< Right motor speed (PWM pin) */

// IR Sensors for Line Following (TCRT5000)
#define LEFT_SENSOR 11   /**< Left IR sensor digital input */
#define RIGHT_SENSOR 12  /**< Right IR sensor digital input */

// ISD1820 Voice Module and Headlights
#define ISD_TRIGGER A3   /**< Voice module trigger pin */
#define HEADLIGHT1 A1    /**< Left headlight LED */
#define HEADLIGHT2 A2    /**< Right headlight LED */

// Ultrasonic Sensor (HC-SR04)
#define TRIG_PIN 9       /**< Ultrasonic trigger pin */
#define ECHO_PIN 10      /**< Ultrasonic echo pin */

// Gas Sensor (MQ-2)
#define GAS_SENSOR A0    /**< MQ-2 gas sensor analog output */

// Sound Sensor (KY-037)
#define SOUND_SENSOR A5  /**< KY-037 sound sensor analog output */

// Buzzer
#define BUZZER A4        /**< Active buzzer control pin */

// ============================================
//                          CONSTANTS
// ============================================

// Motor Speed Constants
#define SPEED_NORMAL 200  /**< Normal driving speed (0-255) */
#define SPEED_TURN 180    /**< Turning speed (0-255) */

// Line Following Speeds (slower for better tracking)
#define LINE_SPEED_FORWARD 90  /**< Forward speed in line-following mode */
#define LINE_SPEED_TURN 70     /**< Turning speed in line-following mode */

// Ultrasonic Constants
#define OBSTACLE_DISTANCE 30   /**< Distance threshold for obstacle detection (cm) */

// Gas Sensor Constants
#define GAS_THRESHOLD 300       /**< Analog threshold for gas detection */
#define GAS_ALARM_DURATION 5000 /**< Alarm duration (ms) */
#define GAS_BLINK_INTERVAL 200  /**< LED blink interval (ms) */

// Sound Sensor Constants
#define SOUND_THRESHOLD 100     /**< Analog threshold for sound detection */
#define SOUND_ALARM_DURATION 5000 /**< Alarm duration (ms) */
#define SOUND_BLINK_INTERVAL 200  /**< LED blink interval (ms) */

// ============================================
//                          GLOBAL VARIABLES
// ============================================

// Mode Selection
bool lineFollowingMode = false;  /**< false = Manual, true = Line Following */

// Ultrasonic Variables
bool emergencyStopActive = false;    /**< Flag for obstacle emergency stop */
unsigned long lastObstacleBlink = 0; /**< Last blink time for obstacle alert */
bool obstacleBeepState = false;      /**< Toggle state for buzzer/headlights */

// Gas Sensor Variables
bool gasDetected = false;            /**< Gas detection flag */
unsigned long gasAlarmStartTime = 0; /**< When gas alarm started */
unsigned long gasLastBlink = 0;      /**< Last blink time for gas alert */
bool gasBlinkState = false;          /**< Toggle state for gas alarm */

// Sound Sensor Variables
bool soundDetected = false;          /**< Sound detection flag */
unsigned long soundAlarmStartTime = 0; /**< When sound alarm started */
unsigned long soundLastBlink = 0;    /**< Last blink time for sound alert */
bool soundBlinkState = false;        /**< Toggle state for sound alarm */

// Headlight State
bool headlightsWereOn = false;       /**< Store previous headlight state */

// Line Following Timing
unsigned long lineLastBlink = 0;     /**< Last blink time for line lost alert */
bool lineBlinkState = false;         /**< Toggle state for line lost alert */

// ============================================
//                          SETUP
// ============================================

/**
 * @brief Arduino setup function - Initializes all pins and peripherals
 * 
 * This function configures all I/O pins, initializes serial communication,
 * performs a startup sequence, and displays system information.
 */
void setup() {
  Serial.begin(9600);  /**< Initialize serial communication at 9600 baud */

  // Motor Control Pins
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  pinMode(ENB, OUTPUT);

  // IR Sensor Pins
  pinMode(LEFT_SENSOR, INPUT);
  pinMode(RIGHT_SENSOR, INPUT);

  // ISD1820 and Headlight Pins
  pinMode(ISD_TRIGGER, OUTPUT);
  pinMode(HEADLIGHT1, OUTPUT);
  pinMode(HEADLIGHT2, OUTPUT);

  // Ultrasonic Sensor Pins
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  // Gas and Sound Sensor Pins
  pinMode(GAS_SENSOR, INPUT);
  pinMode(SOUND_SENSOR, INPUT);

  // Buzzer Pin
  pinMode(BUZZER, OUTPUT);
  digitalWrite(BUZZER, LOW);

  // Ensure all outputs start in OFF state
  digitalWrite(ISD_TRIGGER, LOW);
  digitalWrite(HEADLIGHT1, LOW);
  digitalWrite(HEADLIGHT2, LOW);

  // Stop motors initially
  stopMotors();

  // Perform startup sequence
  startupSequence();

  // Signal ready and display system information
  Serial.println("=========================================");
  Serial.println("GUARDIAN ROBOT - FULLY INTEGRATED");
  Serial.println("=========================================");
  Serial.println("Commands: F, B, L, R, S (Manual Mode)");
  Serial.println(" M (Toggle Manual/Line Following Mode)");
  Serial.println(" Y (Horn), X (Light ON), x (Light OFF)");
  Serial.println("=========================================");
  Serial.println("Current Mode: MANUAL");
  Serial.println("=========================================");
}

// ============================================
//                          MAIN LOOP
// ============================================

/**
 * @brief Main program loop - Executes continuously
 * 
 * The loop follows a priority-based execution:
 * 1. Gas Sensor (HIGHEST PRIORITY)
 * 2. Sound Sensor (SECOND PRIORITY)
 * 3. Ultrasonic Sensor
 * 4. Bluetooth Command Processing
 * 5. Line Following Mode (if active)
 */
void loop() {
  // 1. Check GAS SENSOR (HIGHEST PRIORITY)
  checkGasSensor();
  if (gasDetected) {
    handleGasAlarm();
    return;  // Gas alarm takes absolute priority
  }

  // 2. Check SOUND SENSOR (SECOND PRIORITY)
  checkSoundSensor();
  if (soundDetected) {
    handleSoundAlarm();
    return;  // Sound alarm takes priority over other functions
  }

  // 3. Check ultrasonic sensor for obstacles
  checkObstacle();

  // 4. Check Bluetooth commands
  if (Serial.available() > 0) {
    char command = Serial.read();
    processCommand(command);
  }

  // 5. If in Line Following Mode, run line follower (if no obstacle)
  if (lineFollowingMode && !emergencyStopActive) {
    runLineFollower();
  }

  delay(50);  /**< Small delay to prevent excessive CPU usage */
}

// ============================================
//                          COMMAND PROCESSING
// ============================================

/**
 * @brief Process Bluetooth commands
 * @param command Character command received via serial
 * 
 * Supported Commands:
 * - M: Toggle between Manual and Line Following modes
 * - F: Move forward (Manual mode only)
 * - B: Move backward (Manual mode only)
 * - L: Turn left (Manual mode only)
 * - R: Turn right (Manual mode only)
 * - S: Stop motors
 * - Y: Trigger voice horn + buzzer
 * - X: Turn headlights ON
 * - x: Turn headlights OFF
 */
void processCommand(char command) {
  switch(command) {
    // Mode Toggle
    case 'M': 
    case 'm':
      lineFollowingMode = !lineFollowingMode;
      stopMotors();
      if (lineFollowingMode) {
        Serial.println("MODE: LINE FOLLOWING AUTO");
        Serial.println("Place robot on black line track");
      } else {
        Serial.println("MODE: MANUAL CONTROL");
        Serial.println("Use F, B, L, R, S to control");
      }
      break;

    // Car Control (only in Manual Mode)
    case 'F': 
    case 'f':
      if (!lineFollowingMode && !emergencyStopActive) {
        moveForward();
        Serial.println("FORWARD");
      } else if (lineFollowingMode) {
        Serial.println("In Line Following Mode - Press M for Manual");
      } else {
        Serial.println("OBSTACLE DETECTED - CANNOT MOVE FORWARD");
      }
      break;

    case 'B': 
    case 'b':
      if (!lineFollowingMode) {
        moveBackward();
        Serial.println("BACKWARD");
      } else {
        Serial.println("In Line Following Mode - Press M for Manual");
      }
      break;

    case 'L': 
    case 'l':
      if (!lineFollowingMode) {
        turnLeft();
        Serial.println("LEFT");
      } else {
        Serial.println("In Line Following Mode - Press M for Manual");
      }
      break;

    case 'R': 
    case 'r':
      if (!lineFollowingMode) {
        turnRight();
        Serial.println("RIGHT");
      } else {
        Serial.println("In Line Following Mode - Press M for Manual");
      }
      break;

    case 'S': 
    case 's':
      stopMotors();
      emergencyStopActive = false;
      Serial.println("STOP");
      break;

    // Voice Horn + Buzzer
    case 'Y': 
    case 'y':
      triggerVoiceAndBuzzer();
      Serial.println("HORN");
      break;

    // Headlight Commands
    case 'X':
      digitalWrite(HEADLIGHT1, HIGH);
      digitalWrite(HEADLIGHT2, HIGH);
      headlightsWereOn = true;
      Serial.println("HEADLIGHTS ON");
      break;

    case 'x':
      digitalWrite(HEADLIGHT1, LOW);
      digitalWrite(HEADLIGHT2, LOW);
      headlightsWereOn = false;
      Serial.println("HEADLIGHTS OFF");
      break;

    default:
      break;  // Ignore unknown commands
  }
}

// ============================================
//                          LINE FOLLOWING MODE
// ============================================

/**
 * @brief Execute line following algorithm
 * 
 * Uses two IR sensors to detect a black line on a white surface.
 * The robot adjusts its direction to stay on the line.
 * 
 * Sensor Logic:
 * - Both on BLACK: Go forward
 * - Left on BLACK, Right on WHITE: Turn left
 * - Left on WHITE, Right on BLACK: Turn right
 * - Both on WHITE: Line lost, stop and alert
 */
void runLineFollower() {
  int left = digitalRead(LEFT_SENSOR);
  int right = digitalRead(RIGHT_SENSOR);

  // Both sensors on BLACK line -> GO FORWARD
  if (left == 1 && right == 1) {
    lineForward();
    digitalWrite(HEADLIGHT2, HIGH);  // Green LED
    digitalWrite(HEADLIGHT1, LOW);   // Red OFF
    noTone(BUZZER);
  }
  // Left on BLACK, Right on WHITE -> TURN LEFT
  else if (left == 1 && right == 0) {
    lineTurnLeft();
    digitalWrite(HEADLIGHT1, HIGH);
    digitalWrite(HEADLIGHT2, LOW);
    tone(BUZZER, 800, 30);  // Short beep for left turn
  }
  // Right on BLACK, Left on WHITE -> TURN RIGHT
  else if (left == 0 && right == 1) {
    lineTurnRight();
    digitalWrite(HEADLIGHT1, HIGH);
    digitalWrite(HEADLIGHT2, LOW);
    tone(BUZZER, 1000, 30);  // Short beep for right turn
  }
  // Both sensors on WHITE -> LINE LOST -> STOP
  else {
    stopMotors();
    
    // Blink LEDs to indicate line lost
    unsigned long currentTime = millis();
    if (currentTime - lineLastBlink > 300) {
      lineLastBlink = currentTime;
      lineBlinkState = !lineBlinkState;
      digitalWrite(HEADLIGHT1, lineBlinkState);
      digitalWrite(HEADLIGHT2, lineBlinkState);
    }
    
    // Beep to indicate line lost
    static unsigned long lastBeep = 0;
    if (currentTime - lastBeep > 500) {
      tone(BUZZER, 600, 100);
      lastBeep = currentTime;
    }
  }
}

// ============================================
//                          LINE FOLLOWING MOTOR FUNCTIONS
// ============================================

/**
 * @brief Move forward in line following mode
 */
void lineForward() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  analogWrite(ENA, LINE_SPEED_FORWARD);
  analogWrite(ENB, LINE_SPEED_FORWARD);
}

/**
 * @brief Turn left in line following mode
 */
void lineTurnLeft() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  analogWrite(ENA, 0);
  analogWrite(ENB, LINE_SPEED_TURN);
}

/**
 * @brief Turn right in line following mode
 */
void lineTurnRight() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
  analogWrite(ENA, LINE_SPEED_TURN);
  analogWrite(ENB, 0);
}

// ============================================
//                          GAS SENSOR FUNCTIONS
// ============================================

/**
 * @brief Check gas sensor for detection
 * 
 * Reads the MQ-2 gas sensor and triggers alarm if threshold is exceeded.
 * The gas sensor has the highest priority in the system.
 */
void checkGasSensor() {
  int gasLevel = analogRead(GAS_SENSOR);
  
  // Periodic status print (every 3 seconds)
  static unsigned long lastGasPrint = 0;
  if (millis() - lastGasPrint > 3000) {
    Serial.print("Gas Level (A0): ");
    Serial.println(gasLevel);
    lastGasPrint = millis();
  }

  // Trigger alarm if threshold exceeded
  if (!gasDetected && gasLevel > GAS_THRESHOLD) {
    activateGasAlarm(gasLevel);
  }
}

/**
 * @brief Activate gas alarm
 * @param gasLevel Current gas sensor reading
 * 
 * Stops motors, displays alert message, triggers voice message,
 * and starts visual/audible alarm sequence.
 */
void activateGasAlarm(int gasLevel) {
  gasDetected = true;
  gasAlarmStartTime = millis();
  
  // Emergency stop
  stopMotors();
  
  // Display alert
  Serial.println("=========================================");
  Serial.print(" GAS LEAK DETECTED! ");
  Serial.print(" | Level: ");
  Serial.println(gasLevel);
  Serial.println("=========================================");
  
  // Trigger voice alert
  triggerVoiceMessage();
  
  // Initial flash indication
  digitalWrite(HEADLIGHT1, HIGH);
  digitalWrite(HEADLIGHT2, HIGH);
  delay(200);
  digitalWrite(HEADLIGHT1, LOW);
  digitalWrite(HEADLIGHT2, LOW);
}

/**
 * @brief Handle active gas alarm
 * 
 * Manages the alarm duration, blinking LEDs, and buzzer.
 * Automatically stops after GAS_ALARM_DURATION.
 */
void handleGasAlarm() {
  unsigned long currentTime = millis();

  // Check if alarm duration has expired
  if (currentTime - gasAlarmStartTime >= GAS_ALARM_DURATION) {
    gasDetected = false;
    digitalWrite(HEADLIGHT1, headlightsWereOn ? HIGH : LOW);
    digitalWrite(HEADLIGHT2, headlightsWereOn ? HIGH : LOW);
    digitalWrite(BUZZER, LOW);
    Serial.println("Gas alarm stopped - System normal");
    return;
  }

  // Blink LEDs and buzz at intervals
  if (currentTime - gasLastBlink >= GAS_BLINK_INTERVAL) {
    gasLastBlink = currentTime;
    gasBlinkState = !gasBlinkState;
    
    digitalWrite(HEADLIGHT1, gasBlinkState);
    digitalWrite(HEADLIGHT2, gasBlinkState);
    digitalWrite(BUZZER, gasBlinkState ? HIGH : LOW);
  }
}

// ============================================
//                          ULTRASONIC SENSOR FUNCTIONS
// ============================================

/**
 * @brief Get distance from ultrasonic sensor
 * @return Distance in centimeters (0-400), or 999 if no echo
 */
int getDistance() {
  // Send trigger pulse
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  
  // Measure echo pulse duration (timeout after 30ms)
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 999;  // No echo received
  
  // Calculate distance in cm (speed of sound = 343 m/s)
  int distance = duration * 0.034 / 2;
  return constrain(distance, 0, 400);
}

/**
 * @brief Check for obstacles using ultrasonic sensor
 * 
 * Monitors distance to obstacles and triggers emergency stop if
 * an obstacle is detected within OBSTACLE_DISTANCE.
 * Activates buzzer and LED alerts while obstacle is present.
 */
void checkObstacle() {
  int distance = getDistance();
  
  // Check if obstacle is detected
  if (distance > 0 && distance < OBSTACLE_DISTANCE) {
    // Activate emergency stop if not already active
    if (!emergencyStopActive) {
      emergencyStopActive = true;
      stopMotors();
      Serial.print("EMERGENCY STOP! Obstacle at ");
      Serial.print(distance);
      Serial.println(" cm");
      
      // Initial flash
      digitalWrite(HEADLIGHT1, HIGH);
      digitalWrite(HEADLIGHT2, HIGH);
      delay(200);
      digitalWrite(HEADLIGHT1, headlightsWereOn ? HIGH : LOW);
      digitalWrite(HEADLIGHT2, headlightsWereOn ? HIGH : LOW);
    }
    
    // Blink LEDs and buzz while obstacle is present
    unsigned long currentTime = millis();
    if (currentTime - lastObstacleBlink >= 200) {
      lastObstacleBlink = currentTime;
      obstacleBeepState = !obstacleBeepState;
      
      if (obstacleBeepState) {
        digitalWrite(BUZZER, HIGH);
        digitalWrite(HEADLIGHT1, HIGH);
        digitalWrite(HEADLIGHT2, HIGH);
      } else {
        digitalWrite(BUZZER, LOW);
        digitalWrite(HEADLIGHT1, headlightsWereOn ? HIGH : LOW);
        digitalWrite(HEADLIGHT2, headlightsWereOn ? HIGH : LOW);
      }
    }
  } else {
    // Clear emergency stop when obstacle is removed
    if (emergencyStopActive) {
      emergencyStopActive = false;
      digitalWrite(BUZZER, LOW);
      digitalWrite(HEADLIGHT1, headlightsWereOn ? HIGH : LOW);
      digitalWrite(HEADLIGHT2, headlightsWereOn ? HIGH : LOW);
      Serial.println("Path clear - Emergency stop released");
    }
    obstacleBeepState = false;
  }

  // Periodic distance print (every 2 seconds for debugging)
  static unsigned long lastPrint = 0;
  if (millis() - lastPrint > 2000) {
    if (distance < 100) {  // Only print if within range
      Serial.print("Distance: ");
      Serial.print(distance);
      Serial.println(" cm");
    }
    lastPrint = millis();
  }
}

// ============================================
//                          VOICE FUNCTIONS
// ============================================

/**
 * @brief Trigger ISD1820 voice module
 * 
 * Sends a 150ms HIGH pulse to the trigger pin to play a recorded message.
 */
void triggerVoiceMessage() {
  digitalWrite(ISD_TRIGGER, HIGH);
  delay(150);
  digitalWrite(ISD_TRIGGER, LOW);
}

/**
 * @brief Trigger voice horn with buzzer
 * 
 * Activates both the voice module and buzzer simultaneously
 * for a combined alert effect.
 */
void triggerVoiceAndBuzzer() {
  digitalWrite(BUZZER, HIGH);
  digitalWrite(ISD_TRIGGER, HIGH);
  delay(150);
  digitalWrite(ISD_TRIGGER, LOW);
  delay(300);
  digitalWrite(BUZZER, LOW);
}

// ============================================
//                          MOTOR CONTROL FUNCTIONS (Manual Mode)
// ============================================

/**
 * @brief Move robot forward
 */
void moveForward() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  analogWrite(ENA, SPEED_NORMAL);
  analogWrite(ENB, SPEED_NORMAL);
}

/**
 * @brief Move robot backward
 */
void moveBackward() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  analogWrite(ENA, SPEED_NORMAL);
  analogWrite(ENB, SPEED_NORMAL);
}

/**
 * @brief Turn robot left (pivot turn)
 */
void turnLeft() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
  analogWrite(ENA, SPEED_TURN);
  analogWrite(ENB, SPEED_TURN);
}

/**
 * @brief Turn robot right (pivot turn)
 */
void turnRight() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
  analogWrite(ENA, SPEED_TURN);
  analogWrite(ENB, SPEED_TURN);
}

/**
 * @brief Stop all motors
 * 
 * Sets all motor pins to LOW and applies 0 PWM to brake the motors.
 */
void stopMotors() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
  analogWrite(ENA, 0);
  analogWrite(ENB, 0);
}

// ============================================
//                          STARTUP SEQUENCE
// ============================================

/**
 * @brief Perform startup sequence
 * 
 * Tests all output devices (LEDs, buzzer, voice module)
 * and displays initial sensor readings.
 */
void startupSequence() {
  // Flash headlights 3 times
  for (int i = 0; i < 3; i++) {
    digitalWrite(HEADLIGHT1, HIGH);
    digitalWrite(HEADLIGHT2, HIGH);
    delay(200);
    digitalWrite(HEADLIGHT1, LOW);
    digitalWrite(HEADLIGHT2, LOW);
    delay(200);
  }
  
  // Test buzzer
  digitalWrite(BUZZER, HIGH);
